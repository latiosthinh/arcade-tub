        endAngle: number;   // 0 to 2*PI

      }

    - Class `Dial`:
      - Properties:
        - `pointerAngle`: number (radians, 0 to 2*Math.PI)
        - `baseSpeed`: number = 2.0 (rad/s)
        - `speedBoostMultiplier`: number = 2.5

      - Helper: `normalizeAngle(rad: number): number` keeping values in `[0, 2 * Math.PI)`.
      - `isAngleInArc(angle: number, start: number, end: number): boolean`:
        - If `start <= end`: returns `angle >= start && angle <= end`.
        - If `start > end` (wrapping over 0): returns `angle >= start || angle <= end`.
      - `update(dt: number, speedMultiplier = 1.0, isBoosted = false): void`:
        - Updates `pointerAngle = normalizeAngle(pointerAngle + baseSpeed * speedMultiplier * (isBoosted ? speedBoostMultiplier : 1.0) * dt)`.
      - `checkHit(): { hit: boolean; type?: ZoneType }`:
        - Iterates over `zones`. If `isAngleInArc(pointerAngle, zone.startAngle, zone.endAngle)` is true, returns `{ hit: true, type: zone.type }`.
        - Returns `{ hit: false }` if no match.
      - `resetZones(difficultyLevel = 0): void`:
        - Generates 1 yellow zone ('score') and 1 blue zone ('time').
        - Yellow arc size = `Math.max(minZoneArc, initialYellowArc - difficultyLevel * 0.02)`.
        - Blue arc size = `Math.max(minZoneArc, initialBlueArc - difficultyLevel * 0.015)`.
        - Picks random start angles ensuring zones do not overlap (minimum gap of 0.2 rad).

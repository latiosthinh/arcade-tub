export type TrackLane = 'front' | 'back'; // front = yellow track (y: 520), back = blue track (y: 440)

export type FallingItemType =
  | 'crate_small'   // 100 base pts, width 32, height 24
  | 'crate_medium'  // 150 base pts, width 44, height 30
  | 'crate_large'   // 200 base pts, width 56, height 36
  | 'crate_golden'  // 500 base pts, width 40, height 30
  | 'powerup_repair' // Restores cart health (+35 HP)
  | 'powerup_shield' // Magnetic shield prevents wobble tilt fall-off for 10s
  | 'bomb';          // Explodes, dealing 35 HP damage and scattering stacked crates

export interface FallingItem {
  id: string;
  type: FallingItemType;
  lane: TrackLane;
  x: number;
  y: number;
  width: number;
  height: number;
  vy: number;
  alive: boolean;
  basePoints: number;
}

export interface StackedCrate {
  id: string;
  type: FallingItemType;
  width: number;
  height: number;
  basePoints: number;
  offsetX: number; // Horizontal offset relative to cart center
}

export interface StackBankingResult {
  totalPoints: number;
  crateCount: number;
  multiplier: number;
}

export interface ItemCollisionResult {
  hit: boolean;
  item?: FallingItem;
  caught: boolean;
  isBomb: boolean;
  isRepair: boolean;
  isShield: boolean;
}

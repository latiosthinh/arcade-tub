Integration Guide

Option A: Vanilla HTML / Web Component

<script src="https://your-domain.com/embed.js"></script>

<arcade-game
  game="brick-blitz"
  width="100%"
  height="600px">
</arcade-game>

<script>
  document.querySelector('arcade-game').addEventListener('score', (e) => {
    console.log(`Game: ${e.detail.game}, Score: ${e.detail.score}`);
  });
</script>

Option B: JavaScript / TypeScript SDK

import { ArcadeEmbed } from '@youtube-games/playables-adapter';

const embed = new ArcadeEmbed({
  container: '#game-container',
  game: 'type-strike',
  onScore: (score) => console.log('Score reported:', score),
  onGameReady: () => console.log('Game loaded!')
});

// Control game lifecycle
embed.pause();
embed.resume();
embed.destroy();

Option C: Raw Iframe (Zero Dependencies)

<iframe
  src="https://your-domain.com/games/sky-hopper/index.html"
  width="100%"
  height="600"
  frameborder="0"
  allow="autoplay">
</iframe>
